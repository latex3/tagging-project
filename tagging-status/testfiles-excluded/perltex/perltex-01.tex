\perlnewcommand{\hilbertmatrix}[1]{
  my $result = '
\[
\renewcommand{\arraystretch}{1.3}
';
  $result .= '\begin{array}{' . 'c' x $_[0] . "}\n";
  foreach $j (0 .. $_[0]-1) {
    my @row;
    foreach $i (0 .. $_[0]-1) {
      push @row, ($i+$j) ? (sprintf '\frac{1}{%d}', $i+$j+1) : '1';
    }
    $result .= join (' & ', @row) . " \\\\\n";
  }
  $result .= '\end{array}
\]';
  return $result;
}
